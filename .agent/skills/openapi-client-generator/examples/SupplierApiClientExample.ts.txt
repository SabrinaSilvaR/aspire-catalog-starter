import { apiRequest } from '../api-client'
import type { SupplierDto } from './types'

export const SupplierApi = {
  getAll: () => apiRequest<{ items: SupplierDto[] }>('/api/suppliers'),
  
  getById: (id: string) => apiRequest<SupplierDto>(`/api/suppliers/${id}`),
  
  create: (data: { legalName: string; taxId: string; email: string; phone?: string }) =>
    apiRequest<SupplierDto>('/api/suppliers', {
      method: 'POST',
      body: JSON.stringify(data),
    }),
    
  delete: (id: string) =>
    apiRequest<void>(`/api/suppliers/${id}`, {
      method: 'DELETE',
    }),
}
